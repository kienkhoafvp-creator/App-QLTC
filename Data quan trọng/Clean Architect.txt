    │   │   └── inputs/         # Các ô nhập liệu có tính năng phóng to khi bấm vào
    │   └── layouts/            # Layout chính ép tỷ lệ khung hình thành Mobile dọc
    └── app/                    # Cấu trúc Next.js App Router (Routing của Vercel)
